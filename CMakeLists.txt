cmake_minimum_required(VERSION 3.27)
project(make_believe_morph_engine LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(morph_core STATIC engine/src/audio_ring.cpp engine/src/compiled_preset.cpp engine/src/pitch_mapper.cpp engine/src/pitch_stabilizer.cpp engine/src/yin_tracker.cpp)
target_include_directories(morph_core PUBLIC engine/include)

set(RUBBERBAND_SOURCE_DIR "" CACHE PATH "Path to a Rubber Band Library 4 source checkout")
if(RUBBERBAND_SOURCE_DIR)
    if(NOT EXISTS "${RUBBERBAND_SOURCE_DIR}/single/RubberBandSingle.cpp")
        message(FATAL_ERROR "RUBBERBAND_SOURCE_DIR is not a Rubber Band Library 4 source checkout")
    endif()
    add_library(rubberband_single STATIC "${RUBBERBAND_SOURCE_DIR}/single/RubberBandSingle.cpp")
    target_include_directories(rubberband_single PUBLIC "${RUBBERBAND_SOURCE_DIR}")
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(rubberband_single PRIVATE -Wno-error)
    endif()
    add_library(rubberband_morph_engine STATIC engine/src/rubberband_morph_engine.cpp engine/src/realtime_morph_processor.cpp)
    target_include_directories(rubberband_morph_engine PUBLIC engine/include)
    target_link_libraries(rubberband_morph_engine PUBLIC morph_core rubberband_single)
    add_executable(morph_raw engine/src/audio/morph_raw_main.cpp)
    target_link_libraries(morph_raw PRIVATE rubberband_morph_engine)
endif()

include(CTest)
option(BUILD_BENCHMARKS "Build offline DSP timing benchmarks" OFF)
if(BUILD_TESTING)
    add_executable(pitch_mapper_test engine/tests/pitch_mapper_test.cpp)
    target_link_libraries(pitch_mapper_test PRIVATE morph_core)
    add_test(NAME pitch_mapper_test COMMAND pitch_mapper_test)
    add_executable(yin_tracker_test engine/tests/yin_tracker_test.cpp)
    target_link_libraries(yin_tracker_test PRIVATE morph_core)
    add_test(NAME yin_tracker_test COMMAND yin_tracker_test)
    add_executable(pitch_stabilizer_test engine/tests/pitch_stabilizer_test.cpp)
    target_link_libraries(pitch_stabilizer_test PRIVATE morph_core)
    add_test(NAME pitch_stabilizer_test COMMAND pitch_stabilizer_test)
    add_executable(audio_ring_test engine/tests/audio_ring_test.cpp)
    target_link_libraries(audio_ring_test PRIVATE morph_core)
    add_test(NAME audio_ring_test COMMAND audio_ring_test)
    add_executable(compiled_preset_test engine/tests/compiled_preset_test.cpp)
    target_link_libraries(compiled_preset_test PRIVATE morph_core)
    add_test(NAME compiled_preset_test COMMAND compiled_preset_test "${CMAKE_SOURCE_DIR}/fixtures/synthetic/preset.vmp")
    if(TARGET rubberband_morph_engine)
        add_executable(morph_engine_test engine/tests/rubberband_morph_engine_test.cpp)
        target_link_libraries(morph_engine_test PRIVATE rubberband_morph_engine)
        add_test(NAME morph_engine_test COMMAND morph_engine_test)
    endif()
endif()

add_executable(preset_inspector engine/src/preset_inspector_main.cpp)
target_link_libraries(preset_inspector PRIVATE morph_core)

if(BUILD_BENCHMARKS)
    if(NOT TARGET rubberband_morph_engine)
        message(FATAL_ERROR "BUILD_BENCHMARKS requires RUBBERBAND_SOURCE_DIR")
    endif()
    add_executable(offline_dsp_benchmark engine/benchmarks/offline_dsp_benchmark.cpp)
    target_link_libraries(offline_dsp_benchmark PRIVATE rubberband_morph_engine)
endif()

if(WIN32)
    add_library(wasapi_devices STATIC engine/src/audio/wasapi_devices.cpp)
    target_include_directories(wasapi_devices PUBLIC engine/include)
    target_link_libraries(wasapi_devices PUBLIC ole32 uuid)
    add_executable(audio_devices engine/src/audio/audio_devices_main.cpp)
    target_link_libraries(audio_devices PRIVATE wasapi_devices)
    if(TARGET rubberband_morph_engine)
        add_executable(morph_stream engine/src/audio/morph_stream_main.cpp engine/src/audio/wasapi_stream.cpp)
        target_link_libraries(morph_stream PRIVATE wasapi_devices rubberband_morph_engine)
    endif()
endif()
